133 template <typename CommandStatusMessage,
134 typename CommandID>
135 inline void SendCommandStatusMessage(double aSimTime,
136 const uci::type::MessageStateEnum::EnumerationItem& aMessageState,
137 const CommandID& aCommandID)
138 {
139 try
140 {
141 CommandStatusMessage msg(GetComponentParent());
142 msg.Get().setMessageState(aMessageState);
143
144 ISM_OwnerProducerType ownPro(GetComponentParent());
145 ownPro.Get().setValue(ismownpro::type::CVEnumISMOwnerProducerValues::ISMOWNPRO_USA); //Arbitrary for now
146 msg.Get().getSecurityAttributesGroup().getOwnerProducer().push_back(ownPro.Get());
147
148 auto& data = msg.Get().getMessageData();
149 data.setCommandID(aCommandID);
150 data.setState(GetCommandState());
151 SendMessage(aSimTime, msg);
152 PrintMessageSent(aSimTime, "CommandStatus");
153
154 }
155 catch (const uci::base::UCIException& ex)
156 {
157 ut::log::error() << ex.what() << ":" << ex.getErrorCode();
158 }
159 }

179 template <typename CapabilityStatusMessage>
180 inline void UpdateCapabilityStatusMessage(CapabilityStatusMessage* aMessagePtr)
181 {
182 try
183 {
184 auto& data = aMessagePtr->Get().getMessageData();
185 for (size_t i = 0; i < data.getCapabilityStatus().size(); ++i)
186 {
187 auto availability = uci::type::CapabilityAvailabilityEnum::UCI_AVAILABLE;
188 auto& currentAvailability = data.getCapabilityStatus().at(i).getAvailabilityInfo().getAvailability();
189 if (!CanHandleMessage() ||
190 currentAvailability == uci::type::CapabilityAvailabilityEnum::UCI_DISABLED)
191 {
192 availability = uci::type::CapabilityAvailabilityEnum::UCI_DISABLED;
193 }
194 data.getCapabilityStatus().at(i).getAvailabilityInfo().setAvailability(availability);
195 }
196 }
197 catch (const uci::base::UCIException& ex)
198 {
199 ut::log::error() << ex.what() << ":" << ex.getErrorCode();
200 }
201 }

208 inline uci::type::CapabilityAvailabilityEnum::EnumerationItem GetCapabilityAvailability(const CapabilityStatusMessage* aMessagePtr,
209 const size_t aIndex) const
210 {
211 auto availability = uci::type::CapabilityAvailabilityEnum::UCI_AVAILABLE;
212 auto& currentAvailability = aMessagePtr->Get().getMessageData().getCapabilityStatus().at(aIndex).getAvailabilityInfo().getAvailability();
213 if (!CanHandleMessage())
214 {
215 availability = uci::type::CapabilityAvailabilityEnum::UCI_DISABLED;
216 }
217 else
218 {
219 availability = currentAvailability;
220 }
221 return availability;
