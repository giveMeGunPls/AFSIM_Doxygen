155 template< typename CapabilityType,
156 typename CapabilityStatusType>
157 inline void InitializeModeData(CapabilityType& aCapabilityType,
158 CapabilityStatusType& aStatusType,
159 const size_t& aIndex)
160 {
161 try
162 {
163 std::string label = UCI_Util::RemoveAngleBrackets(mSensorPtr->GetPlatform()->GetName() + ":" + mSensorPtr->GetName());
164 auto modeName = mSensorPtr->GetModeName(aIndex);
165 auto it = mCapabilityToUUID.find(modeName);
166 std::string capabilityUUID = GetInterface()->GetConnection()->generateUUID().getValueAsString();
167 bool defaultSet{ false };
168 if (aIndex == 0)
169 {
170 //This should loop 0-1 times
171 for (const auto& uuid : mUUID_ToModeIndex)
172 {
173 if (uuid.second == 0)
174 {
175 defaultSet = true;
176 capabilityUUID = uuid.first;
177 break;
178 }
179 }
180 }
181 if (it != mCapabilityToUUID.end() && !defaultSet)
182 {
183 capabilityUUID = it->second;
184 }
185 aCapabilityType.Get().getCapabilityID().getUUID().setValue(capabilityUUID);
186 aCapabilityType.Get().getCapabilityID().setDescriptiveLabel(label);
187 mSubsystemStatusMessagePtr->Get().getMessageData().getConfiguration().getCapabilityID().push_back(aCapabilityType.Get().getCapabilityID());
188 mUUID_ToModeIndex.emplace(capabilityUUID, aIndex);
189 UCI_CapabilityControlInterfaceEnumType controlInterfaceEnum(GetComponentParent());
190 if (mSensorPtr->GetModeEntry(aIndex)->GetCueMode() == WsfSensor::SlewMode::cSLEW_FIXED)
191 {
192 controlInterfaceEnum.Get().setValue(uci::type::CapabilityControlInterfacesEnum::UCI_AUTO_MDF);
193 }
194 else
195 {
196 controlInterfaceEnum.Get().setValue(uci::type::CapabilityControlInterfacesEnum::UCI_CAPABILITY_COMMAND);
197 }
198 aCapabilityType.Get().getAcceptedInterface().push_back(controlInterfaceEnum.Get());
199
200 aStatusType.Get().setCapabilityID(aCapabilityType.Get().getCapabilityID());
201 aStatusType.Get().getAvailabilityInfo().setAvailability(uci::type::CapabilityAvailabilityEnum::UCI_AVAILABLE);
202
203 UCI_ControlStatusCapabilityControlType capabilityControl(GetComponentParent());
204 capabilityControl.Get().setCapabilityID(aCapabilityType.Get().getCapabilityID());
205 capabilityControl.Get().getAcceptedInterface().push_back(controlInterfaceEnum.Get());
206 capabilityControl.Get().enablePrimaryController();
207 auto& primaryController = capabilityControl.Get().getPrimaryController();
208 auto& serviceID = GetInterface()->GetConnection()->getMyServiceUUID();
209 auto& systemID = GetInterface()->GetConnection()->getMySystemUUID();
210 primaryController.getServiceID().setUUID(serviceID);
211 primaryController.getSystemID().setUUID(systemID);
212 primaryController.getSystemID().setDescriptiveLabel(GetInterface()->GetConnection()->getMySystemLabel());
213 mPrimaryController[capabilityUUID] = Controller(systemID.getValueAsString(), serviceID.getValueAsString());
214 mControlStatusMessagePtr->Get().getMessageData().getControlType().getCapabilityControl().push_back(capabilityControl.Get());
215 }
216 catch (const uci::base::UCIException& ex)
217 {
218 ut::log::error() << ex.what() << ":" << ex.getErrorCode();
219 }
220 }

262 template <typename CapabilityStatusMessage>
263 inline void UpdateCapabilityStatusMessage(CapabilityStatusMessage* aMessagePtr)
264 {
265 try
266 {
267 auto& data = aMessagePtr->Get().getMessageData();
268 for (size_t i = 0; i < data.getCapabilityStatus().size(); ++i)
269 {
270 auto availability = uci::type::CapabilityAvailabilityEnum::UCI_AVAILABLE;
271 auto& currentAvailability = data.getCapabilityStatus().at(i).getAvailabilityInfo().getAvailability();
272 if (!CanHandleMessage() ||
273 currentAvailability == uci::type::CapabilityAvailabilityEnum::UCI_DISABLED)
274 {
275 availability = uci::type::CapabilityAvailabilityEnum::UCI_DISABLED;
276 }

300 template<typename ActivityMessage>
301 inline void SlewComplete(double aSimTime,
302 ActivityMessage* aMessagePtr)
303 {
304 if (mIsCommanded)
305 {
306 mIsCommanded = false;
307 auto& activity = aMessagePtr->Get().getMessageData().getActivity();
308 auto& lastActivity = activity.at(activity.size() - 1);
309 lastActivity.setActivityState(uci::type::ActivityStateEnum::UCI_COMPLETED);
310 SendActivityMessage(aSimTime, uci::type::MessageStateEnum::UCI_UPDATE);
311
312 double eventTime{ aSimTime + mSensorPtr->GetCurrentMode()->GetFrameTime() };
313 size_t platformIndex{ mSensorPtr->GetPlatform()->GetIndex() };
314 auto resetModeEvent = ut::make_unique<UCI_SetDefaultModeEvent>(mSensorPtr,
315 eventTime,
316 platformIndex,
317 mLowestAvailableCapability);
318 GetSimulation().AddEvent(std::move(resetModeEvent));
319 }
