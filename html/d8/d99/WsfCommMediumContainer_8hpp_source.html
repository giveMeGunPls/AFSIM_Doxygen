132public:
133 bool ProcessAddOrEditCommand(UtInput& aInput, PARENT& aParent, bool aIsAdding) override
134 {
135 // This is a wrapper for medium usage. Redirect input back to the container for proper processing.
136 auto containerPtr = wsf::comm::medium::ContainerComponent<PARENT>::Find(aParent);
137 if (containerPtr)
138 {
139 wsf::comm::medium::Container::InputAction action;
140 if (aIsAdding)
141 {
142 action = wsf::comm::medium::Container::cADD;
143 }
144 else
145 {
146 action = wsf::comm::medium::Container::cEDIT;
147 }
148
149 return containerPtr->ProcessContainerInput(aInput, action);
150 }
151
152 return false;
153 }
154
155 bool ProcessDeleteCommand(UtInput& aInput, PARENT& aParent) override
156 {
157 // This is a wrapper for medium usage. Redirect input back to the container for proper processing.
158 auto containerPtr = wsf::comm::medium::ContainerComponent<PARENT>::Find(aParent);
159 if (containerPtr)
160 {
161 auto action = wsf::comm::medium::Container::cDELETE;
162 return containerPtr->ProcessContainerInput(aInput, action);
163 }
164
165 return false;
166 }
167
168 bool PreInitialize(double aSimTime, PARENT& aParent) override
169 {
170 // At the time PreInitialize is called, all input has been parsed.
171 // If user input has not resulted in a viable medium added to the container,
172 // a default instance is required.
173 auto containerPtr = wsf::comm::medium::ContainerComponent<PARENT>::Find(aParent);
174 if (containerPtr && !containerPtr->GetMedium())
175 {
176 return containerPtr->AddDefaultMedium();
177 }
178
179 return true;
180 }

225 static ContainerComponent<PARENT>* Find(const PARENT& aParent)
226 {
227 const auto& components = aParent.GetComponents();
228 auto containerPtr = components.template GetComponent<ContainerComponent<PARENT>>();
229
230 return containerPtr;

233 static ContainerComponent<PARENT>* FindOrCreate(PARENT& aParent, const WsfScenario& aScenario)
234 {
235 auto containerPtr = Find(aParent);
236 if (!containerPtr)
237 {
238 auto clonedContainer = ut::make_unique<ContainerComponent<PARENT>>(aScenario);
239 auto& components = aParent.GetComponents();
240 if (components.AddComponent(clonedContainer.get()))
241 {
242 containerPtr = clonedContainer.release();
243 }
244 }
245 return containerPtr;
246 }
