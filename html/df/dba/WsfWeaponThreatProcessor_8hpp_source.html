80 struct ThreatReportComparitor
81 {
82 ThreatReportComparitor(QuerySort aQuerySort)
83 : mQuerySort(aQuerySort)
84 {
85 }
86
87 bool operator()(const WsfThreatReport& i, const WsfThreatReport& j)
88 {
89 bool greater = i.mPt_max > j.mPt_max;
90 switch (mQuerySort)
91 {
92 case QUERY_SORT_PK_T0:
93 greater = i.mPk_t0 > j.mPk_t0;
94 break;
95 case QUERY_SORT_PT_T0:
96 greater = i.mPt_t0 > j.mPt_t0;
97 break;
98 case QUERY_SORT_PK_MAX:
99 greater = i.mPk_max > j.mPk_max;
100 break;
101 case QUERY_SORT_PT_MAX:
102 greater = i.mPt_max > j.mPt_max;
103 break;
104 }
105 return greater;
106 }

108 QuerySort mQuerySort;
109 };
