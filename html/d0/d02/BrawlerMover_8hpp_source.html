40/******************************************************************************
41 * OOP representation of a Brawler aircraft mover.
42 * Specifically to replicate this call tree:
43 * NOTE: methods that get/set/update common blocks are left out
44 * all values stored in the class member variables now
45 * perfrm()
46 * flyac()
47 * flyefficient() //for straight flight
48 * flacti()
49 * flyac_short() //for short time steps
50 * flact0()
51 * acarea() //computes aircraft reference area
52 * getdt() //timestep for integration of equations of motion
53 * rates() //computes new yaw, pitch, & roll rates
54 * ratmx1() //aero algorithm type 1
55 * gwreq() //converts commanded angular rates
56 * aoa() //computes current alpha
57 * flact1()
58 * xlift() //computes current lift
59 * udrbe() //updates orientation
60 * drag2()
61 * seteng() //computes new engine settings
62 * desaxf() //convert commanded axial forces
63 * deseng() //translates desired forces into engine settings
64 * stenst() //computes actual engine settings from desired
65 * effthr() //computes effective thrust & drag
66 * thrdec() //translate throttle settings into engine settings
67 * fthrst() //thrust
68 * fntthr() //net thrust
69 * udxv() //updates aircraft position/velocity
70 * fflo() //computes current rate of fuel flow
71 * aproj3() //projects a maneuver forward in time for alternative evaluation
72 * allim()
73 * laccel()
74 * prjacc()
75 * spdtoa()
76 * thstoa()
77 * manwnd()
78 ******************************************************************************/
