63 void* GetPointer() { return mData.mPointer; }
64
65 void SetPointer(void* aPointer)
66 {
67 mData.mInteger = 0; // zero bytes not used by pointer for 32-bit systems
68 mData.mPointer = aPointer;

71

74 {
75 aBuff.SerializeBuffer((char*)&mData.mInteger, sizeof(int64_t));
76 }
