49// A basic stream interface to a buffer
50class WSF_UT_EXPORT UtPackStream
51{
52public:
53 UtPackStream() { mBuffer.SetLittleEndian(); }
54 void Write(const void* aDataPtr, int aBytes) { mBuffer.PutRaw((const char*)aDataPtr, aBytes); }
55 void Read(void* aData, int aBytes)
56 {
57 assert((int)mBuffer.GetValidBytes() >= aBytes);
58 mBuffer.GetRaw((char*)aData, aBytes);
59 }
60 void SkipRead(int aBytes)
61 {
62 assert((int)mBuffer.GetValidBytes() >= aBytes);
63 mBuffer.SetGetPos(mBuffer.GetGetPos() + aBytes);
64 }
65
66 UtBuffer mBuffer;
67};
