248template<bool EnableGuard, typename... Args, typename ArgPacker>
249void WsfScriptObserver::AddEvent(const std::string& aEventName,
250 Observer<Args...>& aObserver,
251 const std::string& aDefaultScriptName,
252 const std::string& aExpectedArgumentTypes,
253 const ArgPacker& aArgPacker)
254{
255 auto callback = [=](double aSimTime, Args&&... args)
256 {
257 UtScriptData scriptRetVal;
258 UtScriptDataPacker scriptArgs{GetSimulation().GetApplication().GetScriptTypes()};
259 auto eg = detail::make_event_guard<EnableGuard>(*this);
260 aArgPacker(scriptArgs, std::forward<Args>(args)...);
261 if (!scriptArgs.mDataList.empty() || aExpectedArgumentTypes.empty())
262 { // If expected arguments weren't packed, don't execute scripts
263 ExecuteScriptList(aSimTime, GetScriptList(GetEventIndex(aEventName)), scriptRetVal, scriptArgs);
264 }
265 };
266 AddEventP(aEventName, aObserver.Connect(callback), aDefaultScriptName, aExpectedArgumentTypes);
267}
