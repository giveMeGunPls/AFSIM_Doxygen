114 WsfPProxyUndoValue(const WsfPProxyUndoValue& aSrc)
115 : mPath(aSrc.mPath)
116 , mUndoPtr(aSrc.mUndoPtr)
117 , mReferenceCount(aSrc.mReferenceCount)
118 {

139 if (this == &aRhs)
140 {

142 }

144 mPath = aRhs.mPath;
145 mUndoPtr = aRhs.mUndoPtr;

194 bool MoveToAttr(const WsfPProxyPath& aPath)
195 {

200 return false;

204 }
