58template<typename FIELD, typename DATA_TYPE>
59struct FillerAccessors
60{
61 typedef VariantTypeSelect<DATA_TYPE> VariantSelector;
62 static void Fill(FieldAccessorType& aAccessor)
63 {
64 aAccessor.mFieldGetter = &FillerAccessors<FIELD, DATA_TYPE>::Get;
65 aAccessor.mFieldSetter = &FillerAccessors<FIELD, DATA_TYPE>::Set;
66 aAccessor.mUsingRawData = false;
67 aAccessor.mAccessor = VariantSelector::cACCESSOR_TYPE;
68 }
69 static void Set(void* aField, const FieldAccessorType::Variant& aVal)
70 {
71 FIELD& field = *((FIELD*)aField);
72 field = (typename FIELD::accessor_type)VariantSelector::Select(aVal);
73 }
74 static void Get(void* aField, FieldAccessorType::Variant& aVal)
75 {
76 FIELD& field = *((FIELD*)aField);
77 VariantSelector::Select(aVal) = (typename VariantSelector::variant_type)field;
78 }
79};

196template<typename FIELD>
197MessageAccessorFiller& operator<<(MessageAccessorFiller& aC, FIELD& aField)
198{
199 const auto& type = typeid(FIELD);
200 FieldAccessorType* fieldAccessorType = FieldAccessorType::Find(type);
201 if (!fieldAccessorType)
202 {
203 int id = (int)FieldAccessorType::sFieldAccessorTypes.size();
204 FieldAccessorType& accessor = FieldAccessorType::sFieldAccessorTypes[type];
205 accessor.mId = id;
206 Accessors::Filler<FIELD>::Fill(accessor);
207 accessor.FormatEnumNames();
208 fieldAccessorType = &accessor;
209 }
210 char* msgBeginPtr = ((char*)aC.mMessagePtr);
211 char* fieldPtr = ((char*)&aField);
212 aC.AddField((int)(fieldPtr - msgBeginPtr), fieldAccessorType);
213 return aC;
214}
