192template< typename Type,
193 typename Factory,
194 uci::base::accessorType::AccessorType TypeName >
195class WSF_OMS_UCI_EXPORT UCI_Type
196{
197 public:
198
199 explicit UCI_Type(WsfSimulation* aSimulationPtr);
200 explicit UCI_Type(WsfPlatform* aPlatformPtr);
201 explicit UCI_Type(UCI_Interface& aInterface);
202 UCI_Type(UCI_Interface& aInterface,
203 const Type& aType);
204 UCI_Type(const UCI_Type& aSrc);
205 ~UCI_Type();
206
207 static uci::base::accessorType::AccessorType GetUCI_TypeId() { return TypeName; }
208
210 Type& Get() const { return *mTypePtr; }
211
213 UCI_Interface& GetInterface() const { return *mInterfacePtr; }
214
215 protected:
216
217 UCI_Type() = default;
218
219 UCI_Interface* mInterfacePtr { nullptr };
220 Type* mTypePtr { nullptr };
221
222 private:
223
224 const UCI_FactoryService<Type, Factory>* mTypeFactoryPtr { nullptr };
225};

227template<typename Type, typename Factory, uci::base::accessorType::AccessorType TypeName>
228inline UCI_Type<Type, Factory, TypeName>::UCI_Type(WsfSimulation* aSimulationPtr)
229 : mInterfacePtr(UCI_Util::GetInterface(aSimulationPtr))

268inline UCI_Type<Type, Factory, TypeName>::UCI_Type(const UCI_Type& aSrc)
269 : mInterfacePtr(aSrc.mInterfacePtr),
