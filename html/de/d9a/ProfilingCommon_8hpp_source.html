68inline uint64_t PackVersionInfo(uint16_t aEpoch, uint16_t aMajor, uint16_t aMinor, uint16_t aPatch)
69{
70 const uint64_t epoch = static_cast<uint64_t>(aEpoch);
71 const uint64_t major = static_cast<uint64_t>(aMajor);
72 const uint64_t minor = static_cast<uint64_t>(aMinor);
73 const uint64_t patch = static_cast<uint64_t>(aPatch);
74
75 return (epoch << 48) | (major << 32) | (minor << 16) | patch;
76}

82inline VersionInfo UnpackVersionInfo(const uint64_t packedVersionInfo)
83{
84 constexpr auto epochShift = 48u;
85 constexpr auto majorShift = 32u;
86 constexpr auto minorShift = 16u;
87
88 constexpr uint64_t epochMask = 0xFFFF000000000000ull;
89 constexpr uint64_t majorMask = 0x0000FFFF00000000ull;
90 constexpr uint64_t minorMask = 0x00000000FFFF0000ull;
91 constexpr uint64_t patchMask = 0x000000000000FFFFull;
92
93 VersionInfo versionInfo;
94 versionInfo.mEpoch = static_cast<uint16_t>((packedVersionInfo & epochMask) >> epochShift);
95 versionInfo.mMajor = static_cast<uint16_t>((packedVersionInfo & majorMask) >> majorShift);
96 versionInfo.mMinor = static_cast<uint16_t>((packedVersionInfo & minorMask) >> minorShift);
97 versionInfo.mPatch = static_cast<uint16_t>(packedVersionInfo & patchMask);
98 return versionInfo;
99}
