64// Compare 2 null-terminated strings
65struct UtilCStringCompare
66{
67 bool operator()(const char* aLhs, const char* aRhs) const
68 {
69 while ((*aLhs | *aRhs) != 0)
70 {
71 if (*aLhs < *aRhs)
72 return true;
73 if (*aRhs < *aLhs)
74 return false;
75 ++aLhs;
76 ++aRhs;
77 }
78 return false;
79 }
80};
