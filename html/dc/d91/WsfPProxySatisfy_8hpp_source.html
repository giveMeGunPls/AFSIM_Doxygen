38 std::string StringOrdValue(WsfParseNode* aParentNodePtr, size_t aOrd)
39 {
40 WsfParseNode* childPtr = aParentNodePtr->GetChild(aOrd);
41 if (childPtr)
42 {
43 return childPtr->mValue.Text();
44 }
45 return "";
46 }
47 std::string StringEval(WsfParseNode* aParentNodePtr, WsfParseActionPart& aExpr)
48 {
49 switch (aExpr.mActionPartType)
50 {
51 case WsfParseActionPart::cNODE_REFERENCE:
52 {
53 WsfParseActionNodeReference* ref = (WsfParseActionNodeReference*)&aExpr;
54 if (ref->mValueTypePtr)
55 {
56 return StringOrdValue(aParentNodePtr, ref->mIndex);
57 }
58 return StringOrdValue(aParentNodePtr, ref->mIndex);
59 }
60 break;
61 case WsfParseActionPart::cLITERAL:
62 {
63 WsfParseActionLiteral* lit = (WsfParseActionLiteral*)&aExpr;
64 return lit->mText;
65 }
66 break;
67 default:
68 break;
69 }
70 return "";
71 }

187 enum SatisfyResultEnum
188 {
189 cNOT_SATISFIABLE = 0,
190 cSATISFIABLE = 1,
191 cADDS_VALUE = 2,
192 cCHANGED_TARGET = 4,
193 cSUCCESS = cADDS_VALUE | cSATISFIABLE,
194 cSUCCESS_COMPLETE = cCHANGED_TARGET | cSUCCESS
195 };
196
197 struct SatisfyResult
198 {
199 SatisfyResult(SatisfyResultEnum aResult)
200 : mResult(static_cast<unsigned int>(aResult))
201 {
202 }
203 SatisfyResult()
204 : mResult(cNOT_SATISFIABLE)
205 {
206 }
207 // True if satisfiable
208 operator bool() const { return 0 != (mResult & cSATISFIABLE); }
209 // True if both satisfiable and adds value
210 bool Success() const { return (cSUCCESS & mResult) == cSUCCESS; }
211
212 bool TargetComplete() const { return (cSUCCESS_COMPLETE & mResult) == cSUCCESS_COMPLETE; }
213 // Combine RHS with this
214 SatisfyResult& operator+=(const SatisfyResult& aRhs)
215 {
216 mResult |= aRhs.mResult;
217 return *this;
218 }
219 unsigned int mResult;
220 };
