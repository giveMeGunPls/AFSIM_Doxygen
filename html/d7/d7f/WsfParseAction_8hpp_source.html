29// Stores an address for accessing a value in the proxy.
30// struct members are stored as integer indexes
31// object map keys are stored as C strings.
32class WsfParseActionAddress
33{
34public:
35 enum class EntryType
36 {
37 cSTRING,
38 cROOT,
39 cNODE,
40 cATTRIBUTE
41 };
42
43 class Entry
44 {
45 public:
46 Entry()
47 : Entry(EntryType::cATTRIBUTE)
48 {
49 }
50 Entry(EntryType aType, size_t aIndex = ut::npos, const char* aName = nullptr)
51 : mType(aType)
52 , mIndex(aIndex)
53 , mName(aName)
54 {
55 }
56 Entry(const char* aName)
57 : Entry(EntryType::cSTRING, ut::npos, aName)
58 {
59 }
60
61 bool IsNodeIndex() const { return mType == EntryType::cNODE; }
62 bool IsAttributeIndex() const { return mType == EntryType::cATTRIBUTE; }
63 bool IsString() const { return mType == EntryType::cSTRING; }
64
65 size_t GetNodeIndex() const { return mIndex; }
66
67 EntryType mType;
68 size_t mIndex;
69
70 // This string's memory is handled by WsfParseDefinitions
71 const char* mName;
72 };
73
74 WsfParseActionAddress& operator+=(const WsfParseActionAddress& aRhs)
75 {
76 mAttrIndices.insert(mAttrIndices.end(), aRhs.mAttrIndices.begin(), aRhs.mAttrIndices.end());
77#ifdef WSF_PARSE_DEBUG
78 if (!mDEBUG_Addr.empty())
79 {
80 mDEBUG_Addr += '.';
81 }
82 mDEBUG_Addr += aRhs.mDEBUG_Addr;
83#endif
84 return *this;
85 }
86 WsfParseActionAddress& operator+=(const Entry& aEntry)
87 {
88 mAttrIndices.push_back(aEntry);
89 return *this;
90 }
91 bool IsEmpty() const { return mAttrIndices.empty(); }
92 std::vector<Entry> mAttrIndices;
93#ifdef WSF_PARSE_DEBUG
94 std::string mDEBUG_Addr;
95#endif
96};
