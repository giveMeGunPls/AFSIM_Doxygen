187// XIO representation of a WsfObject.
188class WSF_EXPORT WsfXIO_ObjectInfo
189{
190public:
191 // ObjectType used to be an enum, but it needs to be an int because it is based on component role.
192 using ObjectType = int32_t;
193
194 enum
195 {
196 cPLATFORM_PART = 0x40000000,
197 cARTICULATED_PART = cPLATFORM_PART | 0x20000000,
198 // Allows one to isolate the WSF component role for platform/articulated parts.
199 // It is 23 bits wide, so the role must be less than 8388608 (2^23).
200 cCOMPONENT_ROLE_MASK = 0x007fffff,
201 // Picks up the general type of a platform part.
202 cPART_TYPE_MASK = 0x607fffff,
203 // The 'DERIVED_TYPE' field allows one to uniquely identify a particular type of part within a
204 // category of platform parts. For instance, one could uniquely identify a particular implementation
205 // of a sensor. It is 6 bits wide so it allows for 64 values [0..63]
206 cDERIVED_TYPE_MASK = 0x1f800000,
207 cDERIVED_TYPE_BEGIN = 0x00800000, // The first non-zero value
208 // Start of the
209 cNO_OBJECT = 0,
210 cPLATFORM = cWSF_COMPONENT_PLATFORM,
211 cTRACK_MANAGER = cWSF_COMPONENT_TRACK_MANAGER,
212 cMOVER_PART = cPLATFORM_PART | cWSF_COMPONENT_MOVER,
213 cCOMM_PART = cARTICULATED_PART | cWSF_COMPONENT_COMM,
214 cPROCESSOR_PART = cPLATFORM_PART | cWSF_COMPONENT_PROCESSOR,
215 cSENSOR_PART = cARTICULATED_PART | cWSF_COMPONENT_SENSOR
216 // XIO components may add additional objects.
217 };
218
219 WsfXIO_ObjectInfo(ObjectType aObjectType)
220 : mObjectType(aObjectType)
221 {
222 }
223
224 bool IsPlatform() const { return mObjectType == cPLATFORM; }
225 bool IsTrackManager() const { return mObjectType == cTRACK_MANAGER; }
226 bool IsPlatformPart() const { return (mObjectType & cPLATFORM_PART) != 0; }
227 bool IsArticulatedPart() const { return (mObjectType & cARTICULATED_PART) != 0; }
228 bool IsSensor() const { return (mObjectType & cPART_TYPE_MASK) == cSENSOR_PART; }
229 bool IsComm() const { return (mObjectType & cPART_TYPE_MASK) == cCOMM_PART; }
230 bool IsProcessor() const { return (mObjectType & cPART_TYPE_MASK) == cPROCESSOR_PART; }
231 bool IsMover() const { return (mObjectType & cPART_TYPE_MASK) == cMOVER_PART; }
232 ObjectType GetObjectType() const { return mObjectType; }
233
234 template<typename T>
235 void Serialize(T& aBuff)
236 {
237 aBuff& mObjectType;
238 }
239
240private:
241 WsfXIO_ObjectInfo& operator=(const WsfXIO_ObjectInfo& aRhs) = delete;
242 WsfXIO_ObjectInfo(const WsfXIO_ObjectInfo& aRhs);
243
245 ObjectType mObjectType;
246};
