47 template<typename DerivedComponent>
48 WsfMultiresolutionTypes<DerivedComponent>& GetMultiresolutionTypes() const
49 {
50 const size_t index = mRoleIndexMap.at(cCOMPONENT_ROLE<WsfMultiresolutionPlatformComponent<DerivedComponent>>());
51 auto& types = dynamic_cast<WsfMultiresolutionTypes<DerivedComponent>&>(*GetScenario().GetTypeLists().at(index));
52 return types;
53 }
54
55private:
56 template<typename DerivedComponent>
57 void AddMultiresolutionType()
58 {
59 mRoleIndexMap[cCOMPONENT_ROLE<WsfMultiresolutionPlatformComponent<DerivedComponent>>()] =
60 GetScenario().GetTypeLists().size();
61 GetScenario().AddTypeList(ut::make_unique<WsfMultiresolutionTypes<DerivedComponent>>(GetScenario()));
62 }
63
64 std::unordered_map<int, size_t> mRoleIndexMap;
65};

118template<typename DerivedComponent>
119WsfMultiresolutionTypes<DerivedComponent>& WsfMultiresolutionTypes<DerivedComponent>::Get(WsfScenario& aScenario)
120{
121 auto& registration =
122 dynamic_cast<WsfMultiresolutionTypesRegistration&>(aScenario.GetExtension(wsf::multiresolution::cEXTENSION_NAME));
123 return registration.GetMultiresolutionTypes<DerivedComponent>();
124}
