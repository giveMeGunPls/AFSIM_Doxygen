160 template<typename Callable>
161 static double ContractTowardRoot(Callable aFunction, double aFixed, double aLimit)
162 {
163 if (aFunction(aFixed) >= 0.0)
164 {
165 return aLimit;
166 }
167 double delta = (aLimit - aFixed) / 2.0;
168 double xTest = aFixed + delta;
169 while (aFunction(xTest) > 0.0)
170 {
171 delta /= 2.0;
172 xTest = aFixed + delta;
173 }
174 return xTest + 2.0 * delta;
175 }
