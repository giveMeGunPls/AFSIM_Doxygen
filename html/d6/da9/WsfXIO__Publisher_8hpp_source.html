165 bool GetData(const WsfXIO_PublishKey aKey, T& aData)
166 {
167 bool ok = false;
168 auto i = mReceivedPublications.find(aKey);
169 if (i != mReceivedPublications.end())
170 {
171 GenBuffer io(&i->second.mData[0], static_cast<int>(i->second.mData.size()));
172 io.SetBigEndian();
173 PakI ar(&io);
174 ar& aData;
175 ok = true;
176 }
177 return ok;
178 }
