40// Determine when the application loses and gains focus; send the ApplicationActivated() event
41class ActivationFilter : public QObject
42{
43public:
44 bool mAppHasFocus;
45
46 explicit ActivationFilter(QObject* aParent = nullptr)
47 : QObject(aParent)
48 , mAppHasFocus(true)
49 {
50 }
51
52 bool eventFilter(QObject* obj, QEvent* event) override
53 {
54 if (event->type() == QEvent::FocusOut || event->type() == QEvent::FocusIn)
55 {
56 QFocusEvent* evt = (QFocusEvent*)event;
57 if (evt->reason() == Qt::ActiveWindowFocusReason)
58 {
59 bool hasFocusNow = nullptr != QApplication::activeWindow();
60 if (hasFocusNow != mAppHasFocus)
61 {
62 mAppHasFocus = hasFocusNow;
63 if (mAppHasFocus)
64 {
65 static bool activating = false;
66 if (!activating) // avoid reentering
67 {
68 activating = true;
69 wizSignals->ApplicationActivated();
70 activating = false;
71 }
72 }
73 }
74 }
75 }
76 return QObject::eventFilter(obj, event);
77 }
78};
