74 {
75 // Check if table has implied azimuth symmetry (i.e. - first value is non-negative).
76 float tempAzimuth = aAzimuth;
77 if ((aAzimuth < 0.0F) && (mIV1.Get(0) >= 0.0F))
78 {
79 tempAzimuth = -aAzimuth;
80 }
81 aAzLookup.Lookup(mIV1, tempAzimuth);
82 aElLookup.Lookup(mIV2, aElevation);
83 }
