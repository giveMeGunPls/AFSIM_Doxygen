76template<int BYTE_COUNT>
77class UtPackBitset
78{
79 static constexpr int cBITS_PER_WORD = 32;
80 static constexpr int cWORD_COUNT = (BYTE_COUNT + 3) / 4;
81
82public:
83 UtPackBitset() { memset(mWords, 0, cWORD_COUNT * 4); }
84
85 UtPackBitset(const UtPackBitset<BYTE_COUNT>& aSrc) { memcpy(mWords, aSrc.mWords, cWORD_COUNT * 4); }
86
87 UtPackBitset<BYTE_COUNT>& operator=(const UtPackBitset<BYTE_COUNT>& aRhs)
88 {
89 memcpy(mWords, aRhs.mWords, cWORD_COUNT * 4);
90 return *this;
91 }
92 void set(int aBitIndex, bool aValue)
93 {
94 const int cWORD_INDEX = aBitIndex / cBITS_PER_WORD;
95 const int cBIT = (1u << (aBitIndex % cBITS_PER_WORD));
96 if (aValue)
97 {
98 mWords[cWORD_INDEX] |= cBIT;
99 }
100 else
101 {
102 mWords[cWORD_INDEX] &= ~cBIT;
103 }
104 }
105 void wordOr(unsigned int aMask) { mWords[0] |= aMask; }
106 void wordAnd(unsigned int aMask) { mWords[0] &= aMask; }
107 void wordXor(unsigned int aMask) { mWords[0] ^= aMask; }
108 bool test(int aBitIndex) const
109 {
110 return 0 != (mWords[aBitIndex / cBITS_PER_WORD] & (1u << (aBitIndex % cBITS_PER_WORD)));
111 }
112
113private:
114 unsigned int mWords[cWORD_COUNT];
115};
