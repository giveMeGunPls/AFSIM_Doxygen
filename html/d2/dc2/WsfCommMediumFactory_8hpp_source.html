107 template<typename MEDIUM_TYPE>
108 void AddFactory(std::unique_ptr<Factory<MEDIUM_TYPE>> aFactory)
109 {
110 auto it = mFactoryMap.find(cMEDIUM_ROLE<MEDIUM_TYPE>());
111 if (it != std::end(mFactoryMap))
112 {
113 throw UtException("Attempting to add a factory to a medium type that is already registered.");
114 }
115
116 aFactory->SetScenario(*mScenarioPtr);
117 mFactoryMap[cMEDIUM_ROLE<MEDIUM_TYPE>()] = std::move(aFactory);
118 }
119
120 bool ProcessInput(UtInput& aInput, Container& aParent) const
121 {
122 const auto& validMediums = aParent.GetValidMediums();
123 for (auto mediumIdentifier : validMediums)
124 {
125 auto it = mFactoryMap.find(mediumIdentifier);
126 if (it != std::end(mFactoryMap))
127 {
128 if (it->second->ProcessInput(aInput, aParent))
129 {
130 return true;
131 }
132 }
133 }
134 return false;
135 }
136
137 bool ProcessAddOrEditCommand(UtInput& aInput, Container& aParent, bool aIsAdding) const
138 {
139 const auto& validMediums = aParent.GetValidMediums();
140 for (auto mediumIdentifier : validMediums)
141 {
142 auto it = mFactoryMap.find(mediumIdentifier);
143 if (it != std::end(mFactoryMap))
144 {
145 if (it->second->ProcessAddOrEditCommand(aInput, aParent, aIsAdding))
146 {
147 return true;
148 }
149 }
150 }
151 return false;
152 }
153
154 bool ProcessDeleteCommand(UtInput& aInput, Container& aParent) const
155 {
156 auto command = aInput.GetCommand();
157 if (command == "medium")
158 {
159 std::string type;
160 aInput.ReadValue(type);
161 auto mediumPtr = aParent.GetMedium();
162
163 if (mediumPtr && (mediumPtr->GetType() == type))
164 {
165 auto mediumIdentifier = mediumPtr->GetMediumIdentifier();
166 auto it = mFactoryMap.find(mediumIdentifier);
167 if (it != std::end(mFactoryMap))
168 {
169 return it->second->ProcessDeleteCommand(aInput, aParent);
170 }
171 }
172
173 throw UtInput::BadValue(aInput, "Unknown medium type to delete: " + type);
174 }
175 return false;
176 }
177
178private:
179 WsfScenario* mScenarioPtr{nullptr};
180 FactoryMap mFactoryMap{};
181};
