218class TimeToInputLockoutField : public NormalField<5, NoStatementMax>
219{
220public:
221 typedef int accessor_type;
222 TimeToInputLockoutField& operator=(double aSeconds)
223 {
224 SetTimeToLockout(aSeconds);
225 return *this;
226 }
227 operator int() const { return GetTimeToLockout(); }
228 int GetTimeToLockout() const { return mValue; }
229 void SetTimeToLockout(double aSeconds)
230 {
231 // Convert from time in seconds to TTL range of 0 - 30
232 if (aSeconds <= 2.0)
233 {
234 mValue = 0;
235 }
236 else
237 {
238 // How to calculate the TTL value based on the time in seconds remaining:
239 // Excel formula: ROUND(ROUND((-1+SQRT(ABS(1-(4*(aSeconds * 2 - 2))))/2+1/(aSeconds*aSeconds)),2),0)
240 // C++ compatible formula: FLOOR(FLOOR((-1+SQRT(ABS(1-(4*(aSeconds * 2 - 2))))/2+1/(aSeconds*Seconds))+0.5,1)+0.5,1)
241 mValue =
242 (int)floor((floor((-1 + sqrt(fabs(1 - (4 * (aSeconds * 2 - 2)))) / 2 + 1 / (pow(aSeconds, 2)))) + 0.5) + 0.5);
243 }
244 }
