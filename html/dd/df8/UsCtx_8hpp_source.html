71// Represents a script value during the compilation process
72class UsVal
73{
74public:
75 enum Flags
76 {
77 cLOCAL_VARIABLE = 1,

79 cGLOBAL_VARIABLE = 4,
80 cEXTERN_VARIABLE = 8,
81 cTEMPORARY_VARIABLE = 16
82 // Only for Wizard:
83 ,
84 cPARAMETER_VARIABLE = 32
85 };
86
87 UsVal()
88 : mSym()
89 , mType()
90 , mMethodNamePtr(nullptr)
91 {
92 }
93 UsVal(const UsVal& aRhs)
94 : mSym(aRhs.mSym)
95 , mType(aRhs.mType)
96 , mInitList(aRhs.mInitList)
97 , mMethodNamePtr(aRhs.mMethodNamePtr)
98 {
99 }
100 UsVal& operator=(const UsVal&) = default;
101 bool IsEmpty() const { return mSym == nullptr && mType == nullptr && !mInitList.IsValid(); }
102 Ty* Type() { return mType; }
103 std::string TypeName();
104
105 bool IsAppMethod() const
106 {
107 // Class method using an implied class reference
108 return (mMethodNamePtr != nullptr) && (mType != nullptr);
109 }
110
111 bool IsScript() const
112 {
113 // User-defined script
114 return (mMethodNamePtr == nullptr) && (mType == nullptr) && (mSym != nullptr);
115 }
116
117 // The symbol if this is a symbol, null if not a symbol
118 UsSymbol* mSym;
119 // The type of value. May be null.
120 // In the case of an App method, this is the type of the class containing the method
121 Ty* mType;
122 // If this value is an initializer list, this points to it
123 UtReferencePtr<UsInitializerList> mInitList;
124 // To represent a method, the name is pointed to here,
125 // and the class type is pointed to by mType
126 const char* mMethodNamePtr;
127};
