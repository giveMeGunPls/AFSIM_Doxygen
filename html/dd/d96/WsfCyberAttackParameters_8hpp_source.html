48 template<class T>
49 void GetParameter(const std::string& aEffectName, T*& aParameter) const
50 {
51 if (mScriptParameter)
52 {
53 // Data was provided via usage in the script context. Unpack as UtScriptData.
54 UtScriptData data;
55 GetAuxDataConst().Get(aEffectName, data);
56 aParameter = data.GetPointer()->GetAppObject<T>();
57 }
58 else
59 {
60 // Data was provided programmatically, and can be unpacked directly.
61 GetAuxDataConst().Get(aEffectName, aParameter);
62 }
63 }
64
65private:
66 // The origin of the parameter is important, as data provided in the script context
67 // will have another level of indirection as a UtScriptData object. This value is
68 // set to true when this parameter is constructed in the script context, and subsequent
69 // usages at retrieving the data stored will require unpacking the UtScriptData object.
70 // This is unfortunately required since UtAttribute get methods throw exceptions upon
71 // failure to cast to a particular type instead of simply returning a nullptr.
72 bool mScriptParameter{false};
73};
