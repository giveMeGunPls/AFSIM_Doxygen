21template<class T>
22T* WsfPProxyValue::CastToValue() const
23{
24 if (mTypePtr && mTypePtr->mTypeStoredKind == T::cPROXY_TYPE_ID)
25 {
26 return (T*)GetDataPtr();
27 }
28 return nullptr;
29}

31// implement conversion and assignment to/from basic values
32#define IMPLEMENT_TYPE_OPERATOR(VALUE_TYPE) \
33 inline WsfPProxyValue::operator VALUE_TYPE*() const { return CastToValue<VALUE_TYPE>(); } \
34 inline WsfPProxyValue& WsfPProxyValue::operator=(const VALUE_TYPE& aVal) \
35 { \
36 VALUE_TYPE* vPtr = CastToValue<VALUE_TYPE>(); \
37 if (vPtr) \
38 *vPtr = aVal; \
39 return *this; \
40 }
